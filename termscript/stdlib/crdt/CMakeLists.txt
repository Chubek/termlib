# The CRDT engine also builds independently of Termscript and its dependencies.
cmake_minimum_required(VERSION 3.15)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(TermscriptCRDT VERSION 1.0 LANGUAGES C)
  include(GNUInstallDirs)
  include(CTest)
endif()
add_library(termscript_crdt_core OBJECT rga.c)
target_compile_features(termscript_crdt_core PUBLIC c_std_11)
set_target_properties(termscript_crdt_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(termscript_crdt_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_library(termscript_crdt STATIC $<TARGET_OBJECTS:termscript_crdt_core>)
add_library(termscript::crdt ALIAS termscript_crdt)
target_include_directories(termscript_crdt PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/termscript/stdlib/crdt>)
install(TARGETS termscript_crdt ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES rga.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/termscript/stdlib/crdt)
install(FILES README.md DESTINATION ${CMAKE_INSTALL_DOCDIR}/crdt)
if(BUILD_TESTING)
  add_executable(test_termscript_crdt_core ../../../tests/test_termscript_crdt_core.c)
  target_link_libraries(test_termscript_crdt_core PRIVATE termscript_crdt)
  add_test(NAME termscript_crdt_core COMMAND test_termscript_crdt_core)
  set_tests_properties(termscript_crdt_core PROPERTIES TIMEOUT 30)
  add_executable(test_termscript_crdt_alloc ../../../tests/test_termscript_crdt_alloc.c)
  target_compile_features(test_termscript_crdt_alloc PRIVATE c_std_11)
  add_test(NAME termscript_crdt_alloc COMMAND test_termscript_crdt_alloc)
  set_tests_properties(termscript_crdt_alloc PROPERTIES TIMEOUT 10)
  find_package(Threads QUIET)
  if(Threads_FOUND AND CMAKE_USE_PTHREADS_INIT)
    add_executable(test_termscript_crdt_threads ../../../tests/test_termscript_crdt_threads.c)
    target_link_libraries(test_termscript_crdt_threads PRIVATE termscript_crdt Threads::Threads)
    add_test(NAME termscript_crdt_threads COMMAND test_termscript_crdt_threads)
    set_tests_properties(termscript_crdt_threads PROPERTIES TIMEOUT 10)
  endif()
endif()
